ThemeData agendaTheme(Brightness b) {
  final p = b == Brightness.light ? AgPalette.light : AgPalette.dark;
  final scheme = ColorScheme(
    brightness: b,
    primary: p.brand,          onPrimary: p.onBrand,
    primaryContainer: p.brandTint, onPrimaryContainer: p.ink,
    error: p.serial,           onError: p.onSerial,
    errorContainer: p.serialTint,  onErrorContainer: p.serial,
    surface: p.paper,          onSurface: p.ink,
    onSurfaceVariant: p.ink2,
    surfaceContainer: p.sheet, surfaceContainerLow: p.sheetSunk,
    outline: p.ink,            outlineVariant: p.rule,
    secondary: p.ink,          onSecondary: p.paper,
    shadow: Colors.transparent, surfaceTint: Colors.transparent,
  );
  final text = TextTheme(
    displayLarge: ruqaa(44, 52), displayMedium: ruqaa(32, 40),
    displaySmall: ruqaa(24, 32),
    titleLarge: plex(22, 30, w700), titleMedium: plex(17, 26, w600),
    bodyLarge:  plex(17, 28, w400), bodyMedium:  plex(15, 24, w400),
    labelLarge: plex(14, 20, w600), labelSmall:  plex(12, 18, w500),
  ).apply(bodyColor: p.ink, displayColor: p.ink);

  return ThemeData(
    useMaterial3: true, colorScheme: scheme, textTheme: text,
    scaffoldBackgroundColor: p.paper,
    splashFactory: NoSplash.splashFactory,   // ختم، مش ripple
    filledButtonTheme: FilledButtonThemeData(style: FilledButton.styleFrom(
      minimumSize: const Size(44, 48), elevation: 0,
      shape: const RoundedRectangleBorder(borderRadius: AgRadius.control))),
    inputDecorationTheme: InputDecorationTheme(
      enabledBorder: UnderlineInputBorder(
        borderSide: BorderSide(color: p.ink, width: 1.5)),
      focusedBorder: UnderlineInputBorder(
        borderSide: BorderSide(color: p.brand, width: 2.5)),
      errorBorder: UnderlineInputBorder(
        borderSide: BorderSide(color: p.serial, width: 2.5))),
    cardTheme: CardThemeData(color: p.sheet, elevation: 0,
      shape: RoundedRectangleBorder(side: BorderSide(color: p.ink))),
    navigationBarTheme: NavigationBarThemeData(height: 64,
      indicatorColor: Colors.transparent, backgroundColor: p.paper),
    bottomSheetTheme: BottomSheetThemeData(backgroundColor: p.sheet,
      shape: const RoundedRectangleBorder(borderRadius: AgRadius.sheet)),
    extensions: [AgTokens.of(p)],  // guilloche, serial, amount*, words
  );
}

الجيلوش من المسلسلعلى الويب: style="--gx1:18%;--gy2:-73%;--gr:7px" بيتحسبوا من 001873. في Flutter: GuillochePainter(seed) جوّه RepaintBoundary، وبيتعمله cache كـui.Picture عشان الموبايلات الرخيصة.r = 4 + n%4 · gx1 = d1d2% · gy2 = −d3d4%
قفل الحسابدوسة طويلة AgMotion.hold مع اهتزاز عند 0 و50 و100%. السيرفر بيلغي التوكن ويرجّع العملاء للشركة، وبعدين push لموبايل البيّاع. شاشة «مقفول» بتفضل مكان التطبيق لحد ما يعمل logout.
قبل ما تعمل mergeمفيش hex مكتوب بإيده. كل رقم tnum. كل «متأخر» جنبه الكلمة. كل زرار أيقونة ليه Semantics. الرقعة مش أصغر من 15px. والنقشة ما تظهرش غير على داتا محمية.
